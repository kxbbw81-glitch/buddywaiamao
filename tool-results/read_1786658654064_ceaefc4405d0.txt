    84→// Mini sparkline SVG component - 80x32px
    85→function MiniSparkline({ data, width = 80, height = 32 }: { data: number[]; width?: number; height?: number }) {
    86→  if (data.length < 2) return null
    87→
    88→  const maxVal = Math.max(...data, 1)
    89→  const minVal = Math.min(...data, 0)
    90→  const range = maxVal - minVal || 1
    91→  const padding = 2
    92→
    93→  const points = data.map((val, i) => {
    94→    const x = padding + (i / (data.length - 1)) * (width - padding * 2)
    95→    const y = padding + (1 - (val - minVal) / range) * (height - padding * 2)
    96→    return `${x},${y}`
    97→  })
    98→
    99→  const polylinePoints = points.join(' ')
   100→  // Build area path: line points + close to bottom-right + bottom-left
   101→  const areaPath = `M${points[0]} ${points.map((p) => `L${p}`).join(' ')} L${padding + (width - padding * 2)},${height} L${padding},${height} Z`
   102→
   103→  return (
   104→    <svg width={width} height={height} viewBox={`0 0 ${width} ${height}`} className="overflow-visible">
   105→      <defs>
   106→        <linearGradient id="sparkGradient" x1="0" y1="0" x2="0" y2="1">
   107→          <stop offset="0%" stopColor="#10b981" stopOpacity="0.2" />
   108→          <stop offset="100%" stopColor="#10b981" stopOpacity="0.02" />
   109→        </linearGradient>
   110→      </defs>
   111→      <path d={areaPath} fill="url(#sparkGradient)" />
   112→      <polyline
   113→        points={polylinePoints}
   114→        fill="none"
   115→        stroke="#10b981"
   116→        strokeWidth={2}
   117→        strokeLinecap="round"
   118→        strokeLinejoin="round"
   119→      />
   120→      <circle
   121→        cx={padding + (width - padding * 2)}
   122→        cy={padding + (1 - (data[data.length - 1] - minVal) / range) * (height - padding * 2)}
   123→        r={2.5}
   124→        fill="#10b981"
   125→      />
   126→    </svg>
   127→  )
   128→}
